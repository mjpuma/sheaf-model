\section{Gate 2 --- Headey-clock actions}

\begin{frame}[label=gate2]{Types are slow; actions are fortnightly}
\topic{Governments can keep a sticky type; the export cut itself is chosen every two weeks.}
\matchbanner{GAME\_CLOCK.md; Headey 2011}{dynamic\_policy.py:1--7}

\citet{headey2011} objects that sit on dates, not marketing years: export
restrictions, import surges, stock-release \emph{announcements}. Gate~2
beta implements only the first, as a climatology-relative stock gate.

Characteristic government types (how much a government cares about
domestic food) can be sticky. The decision $\tau_t$ is not. Conflating
them into one annual tax is what \texttt{core.py} still does. That annual
Nash is a TWIST-era prototype. It is not run here.
\bubbles{Knobs below are illustrative types. They do not flip every step.}
{Time: 24-step wheat spine. $\tau_t\in[0,1]$.}
{\citet{headey2011}; \texttt{diagnostics/GAME\_CLOCK.md}.}
\end{frame}

\begin{frame}{The crisis object is a ratio, not an absolute floor}
\topic{A country restricts when its stocks are thin relative to a normal year, not versus a fixed floor.}
\matchbanner{GAME\_CLOCK.md $S_{i,t}/S^{\mathrm{calm}}_{i,t}$}{dynamic\_policy.climatology\_relative\_cuts}

\[
\tau_{i,t}
=
\begin{cases}
\tau^{\mathrm{on}} & \text{if } t\text{ in the shock year and }
S^{\mathrm{open}}_{i,t}/S^{\mathrm{calm}}_{i,t} < r,\\
0 & \text{otherwise.}
\end{cases}
\]

An absolute stock floor fires every hungry season even in climatology
(the harvest calendar already drives the trough below
\texttt{stu\_target}$\times C$). The ratio against the same calendar step
in a normal year is what distinguishes a crisis from a sawtooth.
\bubbles{$r=0.85$, $\tau^{\mathrm{on}}=0.6$: illustrative.}
{$S$ MMT; ratio dimensionless.}
{\texttt{dynamic\_policy.py:128--185}. Two-pass: open path, then cuts.}
\end{frame}

\begin{frame}{Two-pass: open, then cuts}
\topic{First simulate the shock with trade open; then apply cuts from that open stock path.}
The Gate~0 market loop is not rewritten. Pass 1: simulate the shocked
harvest with $\tau\equiv 0$ (open path) and the climatology harvest.
Pass 2: apply $\tau_t$ from the ratio, simulate again.

Only \texttt{exporter} (Russia) gets the synthetic harvest $\times 0.50$.
Kazakhstan's harvest is unchanged. Ukraine is on the market and does not
play.
\bubbles{Two-pass is a protocol, not a parameter.}
{Harvest multiplier dimensionless.}
{\texttt{simulate\_headey}; \texttt{shock\_year\_harvest}.}
\end{frame}

\begin{frame}{Nested-year BR is a diagnostic, not the crisis object}
\topic{A year-level welfare check used to lock the cut intensity --- not the live fortnight rule.}
\leftoverbanner{welfare() year-mean stock vs s\_gov}{dynamic\_policy.py:87--117}

\[
\mathcal{W}
= \underbrace{\sum_{t\in y} p_t X_{i,t}}_{\text{export revenue}}
- \alpha \bigl(\max(0, s^{\mathrm{gov}}-\bar S_{i,y})\bigr)^2.
\]

$s^{\mathrm{gov}}=\mathtt{gov\_stu}\times C_i^{\mathrm{ann}}=18.13$~MMT for
Russia vs Gate~0 competitive safety $7.55$~MMT. World $p$ does not enter
the penalty. Grid $\{0,0.3,0.6,0.9\}$; shock BR interior at $0.6$.

This nested BR is how $\tau^{\mathrm{on}}$ was locked. The live action
rule is the Headey gate, not this annual loop.
\bubbles{$\alpha=12$, \texttt{gov\_stu}=0.48: illustrative.}
{Revenue: (\$/t)$\times$MMT. Penalty: same units after $\alpha$.}
{\texttt{grid\_best\_response}; \texttt{gate2\_beta\_score.csv}.}
\end{frame}

\begin{frame}{What Gate 2 beta does not claim}
\topic{What this beta does not claim: not 2008 who-banned, not a fortnight Nash, not importer scramble.}
\begin{itemize}
\item Not who banned in 2008 or 2010.
\item Not sequential fortnight Nash.
\item Not importer scramble or Japan-style reserve announcements.
\item Not a split domestic vs export price.
\item Not a retune of CropParams or $\sigma$.
\item Ukraine harvest does not cascade onto Russia (Russia is too fat);
      Kazakhstan was locked as the neighbour for that reason. Disclosed.
\end{itemize}
\end{frame}
