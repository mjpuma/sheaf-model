\section{Gate 2 --- Headey-clock actions}

\begin{frame}[label=gate2]{Types are slow; actions are fortnightly}
\topic{Governments can keep a sticky type; the export cut itself is chosen every two weeks.}
\matchbanner{GAME\_CLOCK.md; Headey 2011}{dynamic\_policy.py:1--7}

\citet{headey2011} objects that sit on dates, not marketing years: export
restrictions, import surges, stock-release \emph{announcements}. Gate~2
beta implements only the first, as a climatology-relative stock gate.

Characteristic government types (how much a government cares about
domestic food) can be sticky. The decision $\tau_t$ is not. Conflating
them into one annual tax is what \texttt{core.py} still does. That annual
Nash is a TWIST-era prototype. It is not run here.
\syms{$\tau_t$ export offer cut this step (fraction in $[0,1]$). Types (who plays, buffer, intensity) are slow; $\tau_t$ is fortnightly.}
\bubbles{Knobs below are illustrative types. They do not flip every step.}
{Time: 24-step wheat spine. $\tau_t\in[0,1]$.}
{\citet{headey2011}; \texttt{diagnostics/GAME\_CLOCK.md}.}
\end{frame}

\begin{frame}{The crisis object is a ratio, not an absolute floor}
\topic{A country restricts when its stocks are thin relative to a normal year, not versus a fixed floor.}
\matchbanner{GAME\_CLOCK.md $S_{i,t}/S^{\mathrm{calm}}_{i,t}$}{dynamic\_policy.climatology\_relative\_cuts}

\[
\tau_{i,t}
=
\begin{cases}
\tau^{\mathrm{on}} & \text{if } t\text{ in the shock year and }
S^{\mathrm{open}}_{i,t}/S^{\mathrm{calm}}_{i,t} < r,\\
0 & \text{otherwise.}
\end{cases}
\]

An absolute stock floor would fire every hungry season even in
climatology, because the harvest calendar already drives the trough below
\texttt{stu\_target}$\times C^{\mathrm{ann}}$. Comparing against the
\emph{same calendar step in a normal year} is what separates a crisis from
a seasonal sawtooth.
\syms{$i$ player country, $t$ step \symsep $\tau_{i,t}$ export-offer cut
(fraction in $[0,1]$) \symsep $\tau^{\mathrm{on}}=0.6$ the cut applied when the
gate fires \symsep $S^{\mathrm{open}}_{i,t}$ stocks on the shocked,
trade-\emph{open} pass (MMT) \symsep $S^{\mathrm{calm}}_{i,t}$ stocks on the
climatology pass at the same step (MMT) \symsep $r=0.85$ trigger on the ratio
(dimensionless) \symsep $C^{\mathrm{ann}}$ mean annual use (MMT/year).}
\bubbles{$r=0.85$, $\tau^{\mathrm{on}}=0.6$: illustrative.}
{$S$ MMT; ratio dimensionless.}
{\texttt{dynamic\_policy.py:128--185}. Two-pass: open path, then cuts.}
\end{frame}

\begin{frame}{Method: three forward passes of the Gate~0 map}
\topic{Endogenous cuts are not found by iterating a game. The same market map is run three times and the cut is read off a threshold.}
\matchbanner{README \S8 Gate 2 method}{dynamic\_policy.simulate\_headey:188--202}

\[
S^{\mathrm{calm}}=\mathcal{M}(H^{\mathrm{seas}},\,\tau\equiv0),
\qquad
S^{\mathrm{open}}=\mathcal{M}(H^{\mathrm{shock}},\,\tau\equiv0),
\]
\[
\tau_{i,t}=\tau^{\mathrm{on}}\cdot
\mathbf{1}\bigl\{S^{\mathrm{open}}_{i,t}/S^{\mathrm{calm}}_{i,t}<r\bigr\}
\ \text{ in the shock year},
\qquad
S^{\mathrm{closed}}=\mathcal{M}(H^{\mathrm{shock}},\,\tau).
\]

The Gate~0 market loop is not rewritten --- $\mathcal{M}$ is literally
\texttt{simulate\_prep}. Only \texttt{exporter} (Russia) receives the
synthetic harvest multiplier $0.50$; Kazakhstan's harvest is unchanged;
Ukraine is on the market but does not play.
\syms{$\mathcal{M}$ the Gate~0 forward map, one call per pass \symsep
$H^{\mathrm{seas}}$ climatology harvest and $H^{\mathrm{shock}}$ the shocked
harvest (MMT/step) \symsep $S^{\mathrm{calm}},S^{\mathrm{open}},
S^{\mathrm{closed}}$ end-of-step stocks on the three passes (MMT)
\symsep $\tau_{i,t}$ offer cut (fraction) and $\tau^{\mathrm{on}}=0.6$ its
value when the gate fires \symsep $r=0.85$ trigger on the stock ratio
(dimensionless) \symsep $\mathbf{1}\{\cdot\}$ indicator, $1$ if true else $0$.}
\bubbles{Three passes is a protocol, not a parameter. No iterated best response.}
{Harvest multiplier dimensionless; $S$ in MMT.}
{\texttt{simulate\_headey}; \texttt{climatology\_relative\_cuts};
\texttt{shock\_year\_harvest}.}
\end{frame}

\begin{frame}{Nested-year BR is a diagnostic, not the crisis object}
\topic{A year-level welfare check used to lock the cut intensity --- not the live fortnight rule.}
\leftoverbanner{welfare() year-mean stock vs s\_gov}{dynamic\_policy.py:87--117}

\[
\mathcal{W}
= \underbrace{\sum_{t\in y} p_t X_{i,t}}_{\text{export revenue}}
- \alpha \bigl(\max(0, s^{\mathrm{gov}}-\bar S_{i,y})\bigr)^2.
\]

$s^{\mathrm{gov}}=\mathtt{gov\_stu}\times C_i^{\mathrm{ann}}=18.13$~MMT for
Russia, against Gate~0 competitive safety of $7.55$~MMT. The world price
does \emph{not} enter the penalty. Grid $\{0,0.3,0.6,0.9\}$; the shock
best response is interior at $0.6$.

This nested best response is how $\tau^{\mathrm{on}}$ was locked. The live
action rule is the Headey gate, not this annual loop.
\syms{$\mathcal{W}$ one year's welfare for the exporter ((\$/t)$\times$MMT)
\symsep $p_t$ world price (real 2010 \$/t) \symsep $X_{i,t}$ tonnes exported
(MMT/step) \symsep $\bar S_{i,y}$ mean stock over the 24 steps of year $y$ (MMT)
\symsep $s^{\mathrm{gov}}$ illustrative government buffer target (MMT)
\symsep $\alpha=\mathtt{fs\_stock\_weight}=12$ penalty weight
((\$/t)$\times$MMT per MMT$^2$) \symsep $(\cdot)_+$ positive part. This is a
diagnostic, not the live $\tau_t$ rule.}
\bubbles{$\alpha=12$, \texttt{gov\_stu}=0.48: illustrative.}
{Revenue: (\$/t)$\times$MMT. Penalty: same units after $\alpha$.}
{\texttt{grid\_best\_response}; \texttt{gate2\_beta\_score.csv}.}
\end{frame}

\begin{frame}{What Gate 2 beta does not claim}
\topic{What this beta does not claim: not 2008 who-banned, not a fortnight Nash, not importer scramble.}
\begin{itemize}
\item Not who banned in 2008 or 2010.
\item Not sequential fortnight Nash.
\item Not importer scramble or Japan-style reserve announcements.
\item Not a split domestic vs export price.
\item Not a retune of CropParams or $\sigma$.
\item Ukraine harvest does not cascade onto Russia (Russia is too fat);
      Kazakhstan was locked as the neighbour for that reason. Disclosed.
\end{itemize}
\syms{$\tau_t$ fortnight offer cut (fraction). Not a \$/t Nash tax. $\sigma$ unused (substitution off).}
\end{frame}
