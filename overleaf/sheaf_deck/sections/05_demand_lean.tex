\begin{frame}{Flex vs industrial use}
\topic{Use splits into price-sensitive food/feed and inelastic industrial (USA maize ethanol).}
\matchbanner{README \S8 $C^{\mathrm{flex}}, C^{\mathrm{ind}}$}{dynamic\_crop.\_demand\_blocks}

\[
C_{i,y} = C^{\mathrm{flex}}_{i,y} + C^{\mathrm{ind}}_{i,y}.
\]

Industrial (inelastic) is empty except USA maize:

\[
C^{\mathrm{ind}}_{\mathrm{USA,y}}
= \min\Bigl(C_{\mathrm{USA,y}},\;
\max\bigl(0,\; \mathrm{FSI}_{\mathrm{USA,y}}

That residual is the RFS-era FSI surge (77 MMT in 2005 $\to$ 163 MMT in
2010; feed fell). Wheat FSI is flat; rice has no FSI attribute in PSD.
China FSI is not treated as ethanol.

\begin{frame}{Lean gap --- README and code disagree on the window}
\topic{How much extra grain a country wants on hand until the next harvest pulse.}
\divbanner{README \S8: $L=\max(0,\sum_{k=1}^{h_t} C - \sum_{k=1}^{h_t} H^{\mathrm{exp}})$}
{dynamic\_crop.py:585--588}

Code (and the comment at line 584) includes the \textbf{current} step:

\[
L_{i,t}^{\mathrm{code}}
= \max\bigl(0,\;
C^{\mathrm{ahead}}_{i,t}+C_{i,t}
- H^{\mathrm{ahead}}_{i,t}-H^{\mathrm{exp}}_{i,t}\bigr).
\]

That is $\sum_{k=0}^{h_t}$, not $\sum_{k=1}^{h_t}$.
\texttt{rolling\_ahead} sums $t+1,\ldots,t+h_t$; adding $C_t$ and
$H^{\mathrm{exp}}_t$ brings in $k=0$.

On 2006--07 wheat, $\max|L^{\mathrm{code}}-L^{\mathrm{README}}|=15.47$~MMT
($15.9\%$ of $\max L$). The \textbf{code} is internally consistent with
its comment. The README equation is missing the current step.
\bubbles{$L$ is a state. $\lambda$ (next slide) is the only reduced-form speed.}
{MMT of cover through the horizon.}
{Implementation: \texttt{dynamic\_crop.py:585--588}. Claim: README \S8.}
\end{frame}

\begin{frame}{Export offers}
\topic{Export offers: surplus above food and the stock target, then cut by the restriction.}
\matchbanner{README \S8 $O_{i,t}$}{dynamic\_crop.py:596}

\[
O_{i,t}
= \max\bigl(0,\, \mathrm{avail}_{i,t}-d_{i,t}-T_{i,t}\bigr)\,(1-\tau_{i,t}).
\]

AMIS (Gate~0) or Headey $\tau_t$ (Gate~2) multiplies \textbf{offers}, not
cleared shipments. Shipments are then demand-constrained by the Armington
clear --- that is why Russia Aug--Dec 2010 offers go to $\times 0.11$ while
ships only go to $\times 0.79$.
\bubbles{$\tau$ is the policy instrument. Offers, not a tax in the $K$ of \S2.}
{$O$: MMT/step. $\tau$: dimensionless.}
{\texttt{dynamic\_crop.py:596}; AMIS map or \texttt{dynamic\_policy}.}
\end{frame}

\begin{frame}{Soft warehouse clip and stock update}
\topic{How stocks update at the end of the step, including a soft drain of overflow.}
\matchbanner{README \S8 warehouse\_lambda}{dynamic\_crop.py:607--623}

\[
C^{\mathrm{used}}_{i,t}
= \min\bigl(d_{i,t},\, \max(0, \mathrm{avail}-O^{\mathrm{shipped}}+ \mathrm{received})\bigr),
\]
\[
S_{i,t+1}
= \max(0, \mathrm{avail}-O^{\mathrm{shipped}}-C^{\mathrm{used}}+\mathrm{received})
- \lambda_W \max(0, S-W).
\]

$\lambda_W=\mathtt{warehouse\_lambda}=0.08$ (defaults to rebuild $\lambda$).
A one-step hard clip incinerated $\sim 10\%$ of wheat/maize harvest and
starved lean-season offers. Soft clip leaves $W$ as an attractor.
\bubbles{$\lambda_W$ reduced-form, locked to $\lambda$.}
{$S$: MMT of stocks at end of step.}
{\texttt{dynamic\_crop.py:607--623}.}
\end{frame}

\begin{frame}{Opening stocks}
\topic{Where opening stocks come from, and the two-year warm-up before 2006.}
Opening $S_{i,0}$ is PSD ending stocks in \texttt{stock\_seed\_year} (default
2005), clipped to $\max(\mathtt{max\_stu}\,C,\, 1.5 s)$, then optionally
spun up for \texttt{spin\_up\_years=2} on climatology harvest, mean flex,
no industrial, no AMIS.

The spin-up is a numerical convenience so 2006 does not start from a raw
MY-end snapshot. The twin is then simulated from that same spun stock.
\bubbles{\texttt{spin\_up\_years=2} structural protocol.}
{MMT.}
{\texttt{prepare\_crop\_run:759--784}. PSD ending stocks via
\texttt{\_ending\_stocks}.}
