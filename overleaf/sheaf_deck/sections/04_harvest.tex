\section{Harvest forcing}

\begin{frame}{Climatology is the score-window mean seasonal path}
\topic{The no-shock harvest path: average production, spread by the calendar.}
\matchbanner{README \S8}{dynamic\_crop.prepare\_crop\_run:786--787}

\[
H^{\mathrm{seas}}_{i,t}
= \bar H_i \cdot w_{i,t},
\qquad
\bar H_i = \frac{1}{Y}\sum_{y\in\text{score window}} Q_{i,y},
\]

with $w_{i,t}$ the calendar weight for that step (sums to 1 over the year).
The twin uses this path. Treatment harvest is \textbf{not} raw PSD year
totals laid on the same calendar.
\bubbles{$\bar H_i$ is data. $w_{i,t}$ is the calendar.}
{$H$: MMT / step. $\bar H$: MMT / year.}
{USDA PSD country production, score window (default 2006--11).
\texttt{seasonal.harvest\_path}.}
\end{frame}

\begin{frame}{Calendar weights are a triangle, then split in half}
\topic{How a harvest season becomes 24 step weights: a triangle over the harvest months, each month split across two fortnights.}
\matchbanner{README \S8 calendar weights}{seasonal.harvest\_month\_weights}

For a window of $N$ months with peak index $i_{\mathrm{peak}}$ along
that window,

\[
w_m \propto \max\bigl(N-|i_m-i_{\mathrm{peak}}|,1\bigr)
\]

on months in the window, $0$ elsewhere. Each month is then split equally
across its two half-month steps. Weights sum to 1 over the year.
Verified on every node.
\bubbles{Agrimate-style mid-season peak. Not a 2008 fit.}
{Dimensionless shares of annual $Q$.}
{\texttt{seasonal.py:44--82}; README \S8 ``Calendar weights.''}
\end{frame}

\begin{frame}{Treatment harvest is climatology $\times$ LOWESS}
\topic{The shocked harvest: that average path times a trend-relative anomaly.}
\matchbanner{README \S8}{dynamic\_crop.py:325--329, data\_usda.detrend\_anomalies}

\[
H_{i,y}^{\mathrm{ann}} = \bar H_i \cdot (1+a_{i,y}),
\qquad
a_{i,y} = \frac{Y_{i,y}-\hat Y_{i,y}}{\hat Y_{i,y}}.
\]

$\hat Y$ is a per-country LOWESS trend on a padded PSD history
(\texttt{hist\_pad=12} years, \texttt{window\_years=10}, tricube local
linear --- Agrimate's method, hand-rolled in \texttt{\_lowess}, not
\texttt{statsmodels}). \texttt{shock\_mode=full} keeps surpluses.
\bubbles{\texttt{window\_years=10} is Agrimate's choice. Not fit to 2008.}
{Anomaly $a$ dimensionless. $H$ MMT/year then spread to MMT/step.}
{\texttt{data\_usda.detrend\_anomalies}; Agrimate detrending.
\texttt{dynamic\_crop.\_harvest\_anomaly\_scalars}.}
\end{frame}

\begin{frame}{Why not the in-sample mean}
\topic{Why the anomaly is versus a smooth trend, not versus the 2006--11 average.}
\matchbanner{README \S8 ``Why this matters''}{verify\_claims.py check\_lowess}

World wheat 2006 versus the 2006--11 \emph{mean} is $-8.9\%$. Versus
LOWESS it is $-4.4\%$. The README's ``$-9\%$ vs that mean but only $-4\%$
vs LOWESS'' is the same fact.

An in-sample mean is contaminated by post-2008 yield growth. The model then
treats 2006/07 as a crash (false May spike) and under-weights 2010 (Russia
drought sits near a boom-inflated mean). Signed anomalies versus trend
isolate \emph{shocks}, not secular growth.
\bubbles{Forcing identity is structural. Do not replace LOWESS with
in-sample $z$-scores to ``fix'' 2006.}
{Percent of trend, not percent of 2006--11 mean.}
{Reproduced: world PSD wheat, \texttt{detrend\_anomalies} on 1994--2011.}
\end{frame}

\begin{frame}{Foresight harvest is a blend}
\topic{What agents expect this step's harvest to be: a blend of shock and climatology.}
\matchbanner{README \S8 $H^{\mathrm{exp}}$}{dynamic\_crop.py:561--565}

\[
H^{\mathrm{exp}}_{i,t}
= \phi\, H_{i,t} + (1-\phi)\, H^{\mathrm{seas}}_{i,t}.
\]

$\phi$ is reduced-form (\texttt{foresight\_phi}: wheat/rice $0.55$, maize
$0.50$). Agents see a mixture of the realized (shocked) harvest and the
seasonal climatology --- finite foresight, not rational expectations.
\bubbles{$\phi\in[0,1]$, reduced-form, locked in \texttt{CropParams}.
Not estimated on 2008.}
{$H$ in MMT/step.}
{\texttt{CropParams.foresight\_phi}; \texttt{GATE0\_PARAMETERIZATION.md} table.}
\end{frame}

\begin{frame}{Lean horizon is a \emph{world} clock}
\topic{How far ahead countries look when they plan stocks: until the next world harvest pulse.}
\matchbanner{README \S8 $h_t$}{seasonal.steps\_to\_harvest\_pulse}

\[
h_t
=\min\Bigl\{h\ge 1:\ 
\sum_{k=1}^{h} H^{\mathrm{world}}_{t+k}
\ge 0.12\,\overline{H}^{\mathrm{world}}\Bigr\},
\]

capped at 24. $H^{\mathrm{world}}=\sum_i H_i$ so every country shares
the same lean clock (NH summer pulse). Verified range on 2006--07 wheat:
$h_t\in[1,14]$.
\bubbles{\texttt{harvest\_pulse\_frac=0.12} structural. \texttt{MAX\_LEAN\_STEPS=24}.}
{$h_t$ in steps (dimensionless).}
{\texttt{seasonal.py:163--189}. World path = \texttt{H.sum(axis=0)}.}
\end{frame}
