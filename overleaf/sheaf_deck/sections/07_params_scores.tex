\begin{frame}{Disclosed Gate 0 leftovers (not hidden knobs)}
\topic{What still does not match history, and is left as a leftover --- not a retune list.}
These are in \texttt{GATE0\_PARAMETERIZATION.md} \S6. They are not a
retune list.

\begin{itemize}
\item \textbf{Prices.} Wheat 2008 still high ($\times 2.27$ vs $\times 1.82$);
      2006/07 still $\times 1.46$ vs obs $\sim\times 0.95$; maize 2010 a bit
      high ($\times 1.70$ vs $\times 1.44$). Wheat 2010 is production-led
      after LOWESS.
\item \textbf{Stocks.} World MY-end matches; many exporters sit on the
      global safety floor. $\Delta$stock signs $\approx$ coin-flip.
      World ex-China maize $\times 1.49$ (China MY September, world bar August).
\item \textbf{Use.} Consumption flat/down (mean flex $+$ $\varepsilon<0$)
      while PSD use rises (wheat corr $-0.17$, maize $-0.64$, rice $-0.74$).
      $\tau$ cuts offers; shipments are demand-constrained.
\end{itemize}
\syms{$p$ 2010 \$/t \symsep stocks MMT \symsep $\varepsilon$ demand elasticity \symsep $\tau$ offer cut (fraction). Leftovers are not retune targets.}
\end{frame}
