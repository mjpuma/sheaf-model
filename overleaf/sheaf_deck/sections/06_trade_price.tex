\begin{frame}{Fill rates update asks}
\topic{$q_{i,t+1}$: fill, blockage $b_t$, mean-reversion to $p$.}
\matchbanner{README \S8 $q_{i,t+1}$}{dynamic\_crop.py:625--636}

\[
\mathrm{fill}_{i,t}
= \frac{\sum_j \mathrm{ship}_{ij}}{\max(O_{i,t},\epsilon)},
\qquad
b_t
= \frac{\sum_{i,j} S_{ij}\,\tau_{i,t}\, D_{j,t}}{\max(\sum_j D_{j,t},\epsilon)}.
\]

\[
q \leftarrow
(1-\beta)\, q \exp\bigl(\alpha(\mathrm{fill}-\theta)+\alpha_r b\cdot\mathbf{1}_{O>0}\bigr)
+ \beta\, p,
\]

then clipped to $[0.45 p_0,\, 2.8 p_0]$. Sold-out $\Rightarrow$ raise ask;
leftover $\Rightarrow$ cut. $\alpha_r=0$ recovers the own-fill law
(rival term next slide).
\bubbles{$\alpha=0.15$, $\theta=0.70$, $\beta=0.18$: reduced-form, shared, locked.}
{$q$: real 2010 \$/t. fill, $b$: dimensionless.}
{\texttt{dynamic\_crop.py:625--636}.}
\end{frame}

\begin{frame}{Locked grain is not world-accessible}
\topic{$\mathrm{locked}_t$, $\mathrm{free}_t$: withheld surplus is not abundance.}
\matchbanner{README \S8 $\mathrm{locked}_t$}{dynamic\_crop.py:639--643}

\[
\mathrm{locked}_t
= \sum_i \tau_{i,t}\max(0, S_{i,t+1}-T_{i,t}),
\qquad
\mathrm{free}_t
= \sum_i S_{i,t+1} - \sum_i L_{i,t} - \mathrm{locked}_t.
\]

If withheld grain counted as free, a ban would look like abundance once
$\eta$ is symmetric. This is why $\eta$ can be symmetric in surplus and
shortage without a ban looking cheap.
\bubbles{Accounting identity. No extra parameter.}
{MMT of world-accessible free cover.}
{\texttt{dynamic\_crop.py:641--643}.}
\end{frame}

\begin{frame}{Scarcity ratio --- README writes $f$; code builds a shift}
\topic{$r_t=(\mathrm{twin}+\mathrm{shift})/(\mathrm{free}+\mathrm{shift})$. No named $f$.}
\divbanner{README \S8 $r_t=(F^{\mathrm{twin}}+f)/(\mathrm{free}+f)$}{dynamic\_crop.py:661--663}

Code:

\[
\mathrm{shift}
= 0.05\sum_i s_i + \max\bigl(0,\, -\min(\mathrm{free}, F^{\mathrm{twin}})\bigr),
\qquad
r_t = \frac{F^{\mathrm{twin}}+\mathrm{shift}}{\mathrm{free}+\mathrm{shift}}.
\]

$f$ is never a named constant. For wheat, $0.05\sum s_i = 6.17$~MMT. The
second term is zero unless free or twin goes negative. README's $f$ is
undefined; the code is a floor that keeps $r_t$ defined and calm-stable.
\bubbles{The $0.05$ is a numerical floor, not a literature elasticity.}
{$r_t$ dimensionless. shift in MMT.}

\begin{frame}{Scarcity price, blend, smoother}
\topic{$p^{\mathrm{scar}}$, $p^\star=\omega p^{\mathrm{tr}}+(1-\omega)p^{\mathrm{scar}}$, $p_t$ AR.}
\matchbanner{README \S8 $p^{\mathrm{scar}}$, $p^\star$, $p_t$}{dynamic\_crop.py:668--680}

If calm ($\mathrm{free}=F^{\mathrm{twin}}$, $\Delta u=0$, $b=0$): $p^\star=p_0$.
Else

\[
p^{\mathrm{scar}}_t
= p_0\, r_t^{\eta}\,(1+\kappa_u\Delta u_t+\kappa_b b_t),
\]
\[
p^\star_t = \omega\, p^{\mathrm{tr}}_t + (1-\omega)\, p^{\mathrm{scar}}_t,
\qquad
p_t = \rho\, p_{t-1} + (1-\rho)\, p^\star_t,
\]

clipped to $[60, 1200]$ \$/t. $\eta$ is symmetric in surplus and shortage
(muting abundance stored gluts). $p^{\mathrm{tr}}$ is the trade-weighted ask
(or $p_{t-1}$ if no trade).
\bubbles{$\omega,\eta,\kappa_u,\kappa_b,\rho$ reduced-form, crop-specific where tabled, locked. Not crisis-by-crisis.}
{$p$: real 2010 \$/t.}
{\texttt{CropParams}: \texttt{inv\_eta}, \texttt{trade\_w}, \texttt{unmet\_kappa},
\texttt{block\_kappa}, \texttt{smooth}.}
\end{frame}

\begin{frame}{One world price, by construction}
\topic{One $p_t$ per crop. No domestic CPI. Category \textbf{E}.}
There is no split domestic vs export ask. A ban withholds grain as stock
and marks it locked; it does \textbf{not} create a cheaper domestic CPI.
That is why Gate~2 welfare cannot put world $p$ in the food-security term
--- a world-price penalty would punish the government for the spike the
ban itself causes (\texttt{dynamic\_policy.welfare} docstring).

Agrimate has a richer agent set (commercial vs strategic stocks, importer
behaviour). SHEAF's Gate~0 market is ask-dominated with a twin residual.
That is \textbf{E} (philosophy), not a failed SPE.
\bubbles{No extra domestic-price parameter because there is no extra price.}
{One $p_t$ per crop, real 2010 \$/t.}
{\texttt{dynamic\_crop.\_simulate\_window} price update;
\texttt{dynamic\_policy.welfare} comment.}
\end{frame}
