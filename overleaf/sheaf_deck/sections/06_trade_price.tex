\begin{frame}{Armington min-clear}
\topic{Preferred trade: each link ships the minimum of that exporter's offer to that destination and that importer's demand from that source.}
\matchbanner{README \S8 $\mathrm{ship}^0_{ij}$}{dynamic\_crop.\_bilateral\_clear}

\[
\mathrm{ship}^0_{ij}
= \min\bigl(O_{i,t}\tilde A_{ij},\, D_{j,t} S_{ij}\bigr).
\]

Tonnes come from $O$ and $D$ (USDA). $A$ and $S$ are FAOSTAT shares
only. Whatever is left after this min goes to the residual pool (next
slide).
\bubbles{No extra parameter on the preferred links.}
{Shipments: MMT/step.}
{\texttt{dynamic\_crop.py:501--503}.}
\end{frame}

\begin{frame}{Residual pool after Armington}
\topic{After preferred links clear, leftover demand may take a fraction of leftover offers.}
\matchbanner{README \S8 residual pool}{dynamic\_crop.py:505--520}

\[
\mathrm{offer}^{\mathrm{left}}_i
=\max\bigl(0,O_i-\textstyle\sum_j\mathrm{ship}^0_{ij}\bigr),
\quad
\mathrm{demand}^{\mathrm{left}}_j
=\max\bigl(0,D_j-\textstyle\sum_i\mathrm{ship}^0_{ij}\bigr),
\]
\[
\mathrm{take}_j=\nu\,\mathrm{demand}^{\mathrm{left}}_j,
\qquad
\mathrm{fill}
=\min\Bigl(1,\;
\tfrac{\sum_i\mathrm{offer}^{\mathrm{left}}_i}{\sum_j\mathrm{take}_j}\Bigr).
\]

Leftover offers are allocated by leftover-demand weights and columns are
clipped not to exceed $\mathrm{take}\cdot\mathrm{fill}$.
$\nu=0.15$ is residual substitution after Armington, \textbf{not}
Gate~1 $\sigma$.
\bubbles{$\nu=\mathtt{residual\_subst}=0.15$ structural. Do not confuse with $\sigma$.}
{MMT/step.}
{\texttt{dynamic\_crop.py:505--520}; README \S8 Armington clear.}
\end{frame}

\begin{frame}{Fill rates update asks}
\topic{How an exporter's ask price moves: sold-out raises it, leftover grain cuts it.}
\matchbanner{README \S8 $q_{i,t+1}$}{dynamic\_crop.py:625--636}

\[
\mathrm{fill}_{i,t}
= \frac{\sum_j \mathrm{ship}_{ij}}{\max(O_{i,t},\epsilon)},
\qquad
b_t
= \frac{\sum_{i,j} S_{ij}\,\tau_{i,t}\, D_{j,t}}{\max(\sum_j D_{j,t},\epsilon)}.
\]

\[
q \leftarrow
(1-\beta)\, q \exp\bigl(\alpha(\mathrm{fill}-\theta)+\alpha_r b\cdot\mathbf{1}_{O>0}\bigr)
+ \beta\, p,
\]

then clipped to $[0.45 p_0,\, 2.8 p_0]$. Sold-out $\Rightarrow$ raise ask;
leftover $\Rightarrow$ cut. $\alpha_r=0$ recovers the own-fill law
(rival term next slide).
\bubbles{$\alpha=0.15$, $\theta=0.70$, $\beta=0.18$: reduced-form, shared, locked.}
{$q$: real 2010 \$/t. fill, $b$: dimensionless.}
{\texttt{dynamic\_crop.py:625--636}.}
\end{frame}

\begin{frame}{Rival markup $\alpha_r$ is a floor, not a 2008 fit}
\topic{When a preferred exporter is banned, the others mark their asks up a bit.}

$\alpha_r=0$ recovers the own-fill law. Not fit to 2008 peaks.

\begin{frame}{Locked grain is not world-accessible}
\topic{Grain held behind a ban does not count as world-accessible stock.}
\matchbanner{README \S8 $\mathrm{locked}_t$}{dynamic\_crop.py:639--643}

\[
\mathrm{locked}_t
= \sum_i \tau_{i,t}\max(0, S_{i,t+1}-T_{i,t}),
\qquad
\mathrm{free}_t
= \sum_i S_{i,t+1} - \sum_i L_{i,t} - \mathrm{locked}_t.
\]

If withheld grain counted as free, a ban would look like abundance once
$\eta$ is symmetric. This is why $\eta$ can be symmetric in surplus and
shortage without a ban looking cheap.
\bubbles{Accounting identity. No extra parameter.}
{MMT of world-accessible free cover.}
{\texttt{dynamic\_crop.py:641--643}.}
\end{frame}

\begin{frame}{Trade-weighted ask}

if anyone ships; otherwise the incoming $p_t$. This is not a separate

\begin{frame}{Scarcity ratio}
\topic{The scarcity ratio: free world cover versus the no-shock twin, with a floor so the ratio stays defined.}
\matchbanner{README \S8 $\mathrm{shift}_t$, $r_t$}{dynamic\_crop.py:661--663}

\[
\mathrm{shift}_t
= 0.05\sum_i s_i + \max\bigl(0,\, -\min(\mathrm{free}_t, F^{\mathrm{twin}}_t)\bigr),
\]
\[
r_t = \frac{F^{\mathrm{twin}}_t+\mathrm{shift}_t}{\mathrm{free}_t+\mathrm{shift}_t}.
\]

For wheat, $0.05\sum s_i = 6.17$~MMT. The second term is zero unless
free or twin goes negative. The $0.05$ is a numerical floor, not a
literature elasticity.
\bubbles{The $0.05$ is a numerical floor, locked.}
{$r_t$ dimensionless. shift in MMT.}
{\texttt{dynamic\_crop.py:661--663}; README \S8 world price.}
\end{frame}

\begin{frame}{Scarcity price, blend, smoother}
\topic{How the world price is built: scarcity term, blend with trade asks, then smooth.}
\matchbanner{README \S8 $p^{\mathrm{scar}}$, $p^\star$, $p_t$}{dynamic\_crop.py:668--680}

If calm ($\mathrm{free}=F^{\mathrm{twin}}$, $\Delta u=0$, $b=0$): $p^\star=p_0$.
Else

\[
p^{\mathrm{scar}}_t
= p_0\, r_t^{\eta}\,(1+\kappa_u\Delta u_t+\kappa_b b_t),
\]
\[
p^\star_t = \omega\, p^{\mathrm{tr}}_t + (1-\omega)\, p^{\mathrm{scar}}_t,
\qquad
p_t = \rho\, p_{t-1} + (1-\rho)\, p^\star_t,
\]

clipped to $[60, 1200]$ \$/t. $\eta$ is symmetric in surplus and shortage
(muting abundance stored gluts). $p^{\mathrm{tr}}$ as on the previous
identity slide.
\bubbles{$\omega,\eta,\kappa_u,\kappa_b,\rho$ reduced-form, crop-specific where tabled, locked. Not crisis-by-crisis.}
{$p$: real 2010 \$/t.}
{\texttt{CropParams}: \texttt{inv\_eta}, \texttt{trade\_w}, \texttt{unmet\_kappa},
\texttt{block\_kappa}, \texttt{smooth}.}
\end{frame}

\begin{frame}{One world price, by construction}
\topic{There is one world price per grain. A ban does not create a cheaper domestic price.}
There is no split domestic vs export ask. A ban withholds grain as stock
and marks it locked; it does \textbf{not} create a cheaper domestic CPI.
That is why Gate~2 welfare cannot put world $p$ in the food-security term
--- a world-price penalty would punish the government for the spike the
ban itself causes (\texttt{dynamic\_policy.welfare} docstring).

Agrimate has a richer agent set (commercial vs strategic stocks, importer
behaviour). SHEAF's Gate~0 market is ask-dominated with a twin residual.
That is \textbf{E} (philosophy), not a failed SPE.
\bubbles{No extra domestic-price parameter because there is no extra price.}
{One $p_t$ per crop, real 2010 \$/t.}
{\texttt{dynamic\_crop.\_simulate\_window} price update;
\texttt{dynamic\_policy.welfare} comment.}
\end{frame}
